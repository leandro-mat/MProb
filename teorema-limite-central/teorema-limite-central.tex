% --- RESUMO (ABSTRACT) ---
\begin{abstract}
Nestas notas, demonstramos o Teorema do Limite Central para sequências de
variáveis aleatórias independentes e identicamente distribuídas, com
variância positiva e finita. A prova usa funções características, o
Teorema da Continuidade de Lévy e um limite exponencial complexo.
Relacionamos os momentos às derivadas da função característica e
obtemos uma expansão de segunda ordem cujo resto é controlado sem
supor a existência do terceiro momento. Estudamos também uma recíproca
que caracteriza a finitude do segundo momento por diferenças segundas
na origem e apresentamos exercícios sobre esses resultados.
A exposição parte dos manuscritos de aula e do Capítulo 3 de
\cite{MR3930614}.
\end{abstract}

\section{Introdução}\label{sec:introducao}

Nas notas de aula sobre funções características, vimos que a independência
transforma a função característica de uma soma em um produto. O Teorema da
Continuidade de Lévy permite, então, estudar a convergência em distribuição
dessas somas por meio da convergência de suas funções características.
Vamos aplicar esse método à demonstração do Teorema do Limite Central.

Sejam $X_1,X_2,\ldots$ variáveis aleatórias independentes e identicamente
distribuídas, com média $\mu$ e variância $\sigma^2\in(0,\infty)$.
A Lei dos Grandes Números descreve a aproximação de
$S_n/n$ por $\mu$, onde $S_n=X_1+\cdots+X_n$.
O Teorema do Limite Central descreve as flutuações de $S_n$ em torno de
$n\mu$, quando medidas na escala $\sigma\sqrt{n}$. Mais precisamente,
mostraremos que
\[
\frac{S_n-n\mu}{\sigma\sqrt{n}}\Rightarrow Z,
\]
onde $Z$ tem distribuição normal padrão. Assim, para quaisquer
$a,b\in\mathbb{R}$ com $a<b$,
\[
\mathbb{P}\left(
a\leqslant\frac{S_n-n\mu}{\sigma\sqrt{n}}\leqslant b
\right)
\longrightarrow
\frac{1}{\sqrt{2\pi}}\int_a^b e^{-x^2/2}\,dx.
\]
A continuidade da função de distribuição normal permitirá passar da
convergência em distribuição a essa formulação por intervalos.

A demonstração segue a sequência dos manuscritos de aula. Primeiro,
recordamos na \Cref{sec:auxiliares} os resultados sobre funções
características que serão usados. Na \Cref{sec:preliminares-complexos},
provamos o limite exponencial complexo
$(1+c_n/n)^n\to e^c$ quando $c_n\to c$.
A \Cref{sec:momentos} relaciona momentos e derivadas e estabelece uma
expansão de segunda ordem com controle do resto sob a hipótese de segundo
momento finito. Na \Cref{sec:tlc}, combinamos esses resultados para provar
o Teorema do Limite Central. Os exercícios estão reunidos na
\Cref{sec:exercicios}.

\section{Funções características e convergência em distribuição}\label{sec:auxiliares}

Recordamos as definições e os resultados das notas de aula sobre funções
características que serão usados adiante. As demonstrações e os cálculos das
funções características dos exemplos encontram-se naquelas notas.

\subsection{Definições}\label{subsec:aux-definicoes}

\begin{definition}[Função característica]\label{def:func-caracteristica}
Seja $X$ uma variável aleatória real. A \emph{função característica} de $X$ é a função
$\varphi_{X}\colon\mathbb{R}\to\mathbb{C}$ dada por
\[
  \varphi_{X}(t) \;=\; \mathbb{E}\bigl[e^{itX}\bigr],
  \qquad \forall t\in\mathbb{R}.
\]
\end{definition}

\begin{definition}[Convergência em distribuição]\label{def:convergencia-distribuicao}
\label{def:man-conv-dist}
Dizemos que uma sequência $(X_n)_{n\geqslant1}$ de variáveis aleatórias converge
em distribuição para uma variável aleatória $X$, e escrevemos $X_n\Rightarrow X$,
se
\[
F_{X_n}(x)=\mathbb{P}(X_n\leqslant x)
\longrightarrow
F_X(x)=\mathbb{P}(X\leqslant x)
\]
em todo ponto $x\in\mathbb{R}$ em que a função de distribuição $F_X$ é contínua.
\end{definition}

\begin{definition}[Sequência condensada]\label{def:sequencia-condensada}
Dizemos que uma sequência de variáveis aleatórias $(X_n)_{n\geqslant1}$ é
\emph{condensada} se, para todo $\varepsilon>0$, existe $M>0$ tal que
\[
\mathbb{P}(|X_n|>M)<\varepsilon
\qquad\text{para todo }n\geqslant1.
\]
\end{definition}

\subsection{Resultados}\label{subsec:aux-resultados}

\begin{theorem}\label{thm:propriedades-fc}
Seja $X$ uma variável aleatória real, com função característica $\varphi_X$. Para
todos $t,h,a,b\in\mathbb{R}$ valem:
\begin{enumerate}[label=(\alph*)]
\item $\varphi_X(0)=1$;
\item $\varphi_X(-t)=\overline{\varphi_X(t)}$;
\item $|\varphi_X(t)|=\bigl|\mathbb{E}[e^{itX}]\bigr|\leqslant\mathbb{E}\bigl[|e^{itX}|\bigr]=1$;
\item $|\varphi_X(t+h)-\varphi_X(t)|\leqslant\mathbb{E}\bigl[|e^{ihX}-1|\bigr]$.
Em particular, a função $\varphi_X$ é uniformemente contínua em $\mathbb{R}$;
\item $\mathbb{E}\bigl[e^{it(aX+b)}\bigr]=e^{itb}\varphi_X(at)$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Função característica de uma soma independente]
\label{thm:fc-soma-independentes}
Sejam $X_1,\ldots,X_n$ variáveis aleatórias independentes. Então
\[
\varphi_{X_1+\cdots+X_n}(t)
=\prod_{j=1}^n\varphi_{X_j}(t),
\qquad t\in\mathbb{R}.
\]
Em particular, se elas têm a mesma distribuição,
\[
\varphi_{X_1+\cdots+X_n}(t)
=\bigl(\varphi_{X_1}(t)\bigr)^n.
\]
\end{theorem}

\begin{theorem}[Derivação dentro da esperança]\label{thm:derivacao-integral}
Sejam $I\subseteq\mathbb{R}$ um intervalo aberto, $X$ uma variável aleatória
real e $g:I\times\mathbb{R}\to\mathbb{C}$. Suponha que:
\begin{enumerate}
\item[(i)] para cada $t\in I$, a variável aleatória $g(t,X)$ é integrável;
\item[(ii)] para quase todo $\omega\in\Omega$, a função
$t\mapsto g(t,X(\omega))$ é derivável em $I$;
\item[(iii)] existe uma variável aleatória integrável $Y$ tal que, quase
certamente,
\[
\left| \frac{\partial}{\partial t}\, g(t,X)\right|\leqslant Y,
\qquad \forall t\in I.
\]
\end{enumerate}
Então $F(t)=\mathbb{E}\bigl[g(t,X)\bigr]$ é derivável em $I$ e
\[
F'(t)
=
\frac{d}{dt} \mathbb{E}\left[\, g(t,X)\right]
=
\mathbb{E}\left[ \frac{\partial}{\partial t}\, g(t,X)\right]
\qquad \forall t\in I.
\]
\end{theorem}

\begin{lemma}[Cota para as caudas]\label{lem:cota-cauda-fc}
Seja $Y$ uma variável aleatória real com função característica $\varphi_Y$.
Para todo $u>0$, vale a desigualdade
\begin{equation}\label{eq:cota-concentracao}
\mathbb{P}(|Y|\geqslant2/u)
\leqslant
\frac{1}{u}\int_{-u}^{u}
\bigl(1-\operatorname{Re}\varphi_Y(t)\bigr)\,dt.
\end{equation}
\end{lemma}

\begin{theorem}[Teorema da Continuidade de Lévy]\label{thm:continuidade}
Sejam $X_1,X_2,\ldots$ variáveis aleatórias reais, e denote por
$\varphi_{X_n}$ a função característica de $X_n$, para cada $n$.

\begin{enumerate}[label=\textup{(\roman*)}]
\item Se $X_n\Rightarrow X$, então
$\varphi_{X_n}(t)\to\varphi_X(t)$ para todo $t\in\mathbb{R}$.

\item Se $\varphi_{X_n}(t)\to\varphi(t)$ para todo $t\in\mathbb{R}$, para uma
função $\varphi$ contínua em $0$, então existe uma variável aleatória $X$ tal que
$X_n\Rightarrow X$ e $\varphi=\varphi_X$. Além disso, $(X_n)$ é condensada.
\end{enumerate}
\end{theorem}

No item (ii) do teorema, $\varphi$ é inicialmente apenas uma função
limite. Uma das conclusões do teorema é que essa função limite é função
característica de alguma variável aleatória.

\subsection{Exemplos}\label{subsec:aux-exemplos}

\begin{example}[Distribuição de Rademacher]\label{ex:rademacher}
Seja $X$ uma variável aleatória com distribuição de Rademacher, isto é, suponha
que $X$ assuma apenas os valores $1$ e $-1$, com função de probabilidade dada por
$\mathbb{P}(X=1)=\mathbb{P}(X=-1)=\tfrac12$. Então
\[
\varphi_X(t)=\frac{e^{it}+e^{-it}}{2}=\cos t, \qquad t\in\mathbb{R}.
\]
\end{example}

\begin{example}[Distribuição de Cauchy]\label{ex:cauchy}
A função $f(x)=1/\bigl(\pi(1+x^{2})\bigr)$, $x\in\mathbb{R}$, é uma densidade de
probabilidade. Se $X$ tem densidade $f$, sua função característica é
\[
\varphi_X(t)=e^{-|t|}, \qquad t\in\mathbb{R}.
\]
Além disso, $\mathbb{E}\bigl[|X|\bigr]=\infty$.
\end{example}

\begin{example}[Distribuição normal padrão]\label{ex:normal}
Seja $Z$ uma variável aleatória com distribuição normal padrão, isto é,
com densidade
\[
f_Z(x)=\frac{1}{\sqrt{2\pi}}e^{-x^2/2},
\qquad x\in\mathbb{R}.
\]
Nas notas de aula sobre funções características, calculamos
\[
\varphi_Z(t)=e^{-t^2/2},
\qquad t\in\mathbb{R}.
\]
Essa será a função limite que identificaremos na demonstração do
Teorema do Limite Central.
\end{example}

\section{Preliminares de análise complexa}\label{sec:preliminares-complexos}

Na prova do Teorema do Limite Central, a função característica da soma
normalizada terá a forma $(1+c_n/n)^n$, com $c_n$ complexo e convergente.
Vamos provar o limite dessa expressão comparando cada fator $1+c_n/n$
com $e^{c_n/n}$. Para controlar o erro do produto, começamos pelo
seguinte lema.

\begin{lemma}[Diferença de produtos]\label{lem:man-diferenca-produtos}
Sejam $n\geqslant1$ um inteiro e $R>0$. Considere números complexos
$z_1,\ldots,z_n$ e $w_1,\ldots,w_n$ tais que
$|z_j|\leqslant R$ e $|w_j|\leqslant R$ para todo $j=1,\ldots,n$. Então
\[
\left|\prod_{j=1}^{n}z_j - \prod_{j=1}^{n}w_j\right|
\leqslant R^{n-1}\sum_{k=1}^{n}|z_k-w_k|.
\]
\end{lemma}

\begin{proof}
A prova será feita por indução. No caso $n=1$ temos
\[
|z_1-w_1|\leqslant R^{0}|z_1-w_1|,
\]
que é verdadeiro para quaisquer $z_1,w_1\in\mathbb{C}$ com $|z_1|,|w_1|\leqslant R$.
Suponha que a desigualdade seja válida para $m\geqslant 1$. Sejam
$z_1,\ldots,z_{m+1},w_1,\ldots,w_{m+1}\in\mathbb{C}$ com módulo menor ou igual a $R$.
Então
\begin{align*}
\left|\prod_{j=1}^{m+1}z_j - \prod_{j=1}^{m+1}w_j\right|
&= \left|z_1\prod_{j=2}^{m+1}z_j - w_1\prod_{j=2}^{m+1}w_j\right| \\
&\leqslant \left|z_1\prod_{j=2}^{m+1}z_j - z_1\prod_{j=2}^{m+1}w_j\right|
+ \left|z_1\prod_{j=2}^{m+1}w_j - w_1\prod_{j=2}^{m+1}w_j\right| \\
&= |z_1|\left|\prod_{j=2}^{m+1}z_j - \prod_{j=2}^{m+1}w_j\right|
+ \left|\prod_{j=2}^{m+1}w_j\right||z_1-w_1| \\
&\leqslant R\left|\prod_{j=2}^{m+1}z_j - \prod_{j=2}^{m+1}w_j\right| + R^{m}|z_1-w_1| \\
&\leqslant R\cdot R^{m-1}\sum_{j=2}^{m+1}|z_j-w_j| + R^{m}|z_1-w_1|
\qquad\text{(hipótese de indução)} \\
&= R^{m}\sum_{j=1}^{m+1}|z_j-w_j|,
\end{align*}
o que prova o lema.
\end{proof}

Para comparar os fatores $1+c_n/n$ e $e^{c_n/n}$, precisamos de uma
cota para o erro da aproximação de $e^b$ por $1+b$. O próximo lema
mostra que esse erro é de ordem $|b|^2$ quando $|b|\leqslant1$.

\begin{lemma}[Erro da aproximação exponencial]\label{lem:man-exp-quadratica}
Se $b\in\mathbb{C}$ e $|b|\leqslant 1$, então
\[
\bigl|e^{b}-(1+b)\bigr|\leqslant |b|^{2}.
\]
\end{lemma}

\begin{proof}
Temos
\[
e^{b}-(1+b)=\frac{b^{2}}{2!}+\frac{b^{3}}{3!}+\frac{b^{4}}{4!}+\cdots.
\]
Usando a desigualdade triangular e a hipótese $|b|\leqslant 1$, obtemos a
seguinte estimativa:
\begin{align*}
\bigl|e^{b}-(1+b)\bigr|
&\leqslant \frac{|b|^{2}}{2!}+\frac{|b|^{3}}{3!}+\frac{|b|^{4}}{4!}+\cdots \\
&\leqslant \left(\frac{1}{2!}+\frac{1}{3!}+\frac{1}{4!}+\cdots\right)|b|^{2} \\
&\leqslant \frac{|b|^{2}}{2}\left(1+\frac{2}{3!}+\frac{2}{4!}+\cdots\right) \\
&\leqslant \frac{|b|^{2}}{2}\left(1+\frac{1}{3}+\frac{1}{4\cdot 3}+\frac{1}{5\cdot 4\cdot 3}+\cdots\right) \\
&\leqslant \frac{|b|^{2}}{2}\left(1+\frac{1}{2}+\frac{1}{2^{2}}+\frac{1}{2^{3}}+\cdots\right) \\
&= \frac{|b|^{2}}{2}\sum_{j=0}^{\infty}\left(\frac{1}{2}\right)^{j}
= \frac{|b|^{2}}{2}\cdot\frac{1}{1-\frac{1}{2}}
= |b|^{2}. \qedhere
\end{align*}
\end{proof}

\begin{theorem}[Limite exponencial complexo]\label{thm:man-limite-exp}
Se $(c_n)_{n\geqslant1}$ é uma sequência de números complexos tal que
$\lim_{n\to\infty}c_n=c\in\mathbb{C}$, então
\[
\left(1+\frac{c_n}{n}\right)^{n}\xrightarrow{n\to\infty} e^{c}.
\]
\end{theorem}

\begin{proof}
Vamos comparar $(1+c_n/n)^n$ com $e^{c_n}$. O erro entre cada par de
fatores será de ordem $1/n^2$; o lema dos produtos permitirá somar
esses $n$ erros mantendo uma quantidade que tende a zero.

Para cada $n\in\mathbb{N}$ fixado, considere as sequências finitas constantes
de números complexos definidas por
\[
z_j \equiv 1+\frac{c_n}{n}
\qquad\text{e}\qquad
w_j \equiv e^{c_n/n},
\qquad j=1,\ldots,n.
\]
Seja $R>|c|$. Então existe $n_0\in\mathbb{N}$ tal que $|c_n|\leqslant R$ e
$|c_n|/n\leqslant 1$ para todo $n\geqslant n_0$. Já que
$1+\frac{R}{n}\leqslant e^{R/n}$ (pois $1+x\leqslant e^{x}$ para $x\geqslant 0$),
temos $|z_j|\leqslant 1+\frac{|c_n|}{n}\leqslant 1+\frac{R}{n}\leqslant e^{R/n}$ e
$|w_j|=e^{\operatorname{Re}(c_n)/n}\leqslant e^{|c_n|/n}\leqslant e^{R/n}$ para
todo $j=1,\ldots,n$.

Observe que, para $n\geqslant n_0$,
\begin{align*}
\left|\left(1+\frac{c_n}{n}\right)^{n}-e^{c_n}\right|
&= \left|\left(1+\frac{c_n}{n}\right)^{n}-\left(e^{c_n/n}\right)^{n}\right| \\
&= \left|\prod_{j=1}^{n}z_j-\prod_{j=1}^{n}w_j\right| \\
&\leqslant \left(e^{R/n}\right)^{n-1}\sum_{j=1}^{n}|z_j-w_j|
\qquad\text{(\Cref{lem:man-diferenca-produtos})} \\
&= e^{R\frac{n-1}{n}}\sum_{j=1}^{n}\left|1+\frac{c_n}{n}-e^{c_n/n}\right| \\
&\leqslant e^{R\frac{n-1}{n}}\sum_{j=1}^{n}\frac{|c_n|^{2}}{n^{2}}
\qquad\text{(\Cref{lem:man-exp-quadratica})} \\
&= e^{R\frac{n-1}{n}}\frac{|c_n|^{2}}{n}
\leqslant \frac{e^{R\frac{n-1}{n}}R^{2}}{n}
\xrightarrow{n\to\infty} 0.
\end{align*}
Já que $c_n\to c$ (e portanto $e^{c_n}\to e^{c}$ por continuidade da exponencial),
temos que
\[
\left(1+\frac{c_n}{n}\right)^{n}\xrightarrow{n\to\infty} e^{c}. \qedhere
\]
\end{proof}

\section{Momentos e expansão da função característica}\label{sec:momentos}

Seja $X$ uma variável aleatória real. Nas notas de aula sobre momentos e funções
geradoras de momentos, vimos que as esperanças das potências de $X$ podem ser
recuperadas a partir das derivadas de
$M_X$ na origem, quando a função geradora existe em uma vizinhança de $0$.
Vamos investigar uma relação semelhante para a função característica
\[
\varphi_X(t)=\mathbb{E}[e^{itX}].
\]

A função característica está definida para todo $t\in\mathbb{R}$, pois
$|e^{itX}|=1$. Isso não garante, por si só, que ela seja derivável. Por
exemplo, no \Cref{ex:cauchy} encontramos uma variável sem primeiro momento
finito cuja função característica é $e^{-|t|}$, que não é derivável em
$t=0$.

Para entender o papel dos momentos, observamos que, ao derivar a exponencial
em relação a $t$, aparece um fator $X$:
\[
\frac{d}{dt}e^{itX}=iX e^{itX},
\qquad
\left|\frac{d}{dt}e^{itX}\right|=|X|.
\]
Assim, para justificar a passagem da derivada para dentro da esperança, uma
hipótese natural é que $\mathbb{E}[|X|]<\infty$. O resultado abaixo mostra
como o mesmo raciocínio se estende às derivadas de ordem superior.

\subsection{Derivadas, momentos e o decaimento das caudas}\label{subsec:momentos-derivadas}

O \Cref{lem:cota-cauda-fc} fornece uma cota que relaciona a função
característica às probabilidades de valores grandes de $|X|$: para $u>0$,
\[
\mathbb{P}(|X|\geqslant2/u)
\leqslant
\frac{1}{u}\int_{-u}^{u}
\bigl(1-\operatorname{Re}\varphi_X(t)\bigr)\,dt.
\]
Essa é a cota \eqref{eq:cota-concentracao}, escrita para a variável $X$. Ela
mostra que, quando a parte real de $\varphi_X$ está próxima de $1$ perto da
origem, a probabilidade de $|X|$ ser grande é pequena.

Os momentos também controlam essas probabilidades. De fato, se
$\mathbb{E}[|X|^n]<\infty$ para algum inteiro $n\geqslant1$, então, para
$r>0$,
\[
\mathds{1}_{\{|X|\geqslant r\}}\leqslant\frac{|X|^n}{r^n}.
\]
Tomando esperanças, obtemos
\[
\mathbb{P}(|X|\geqslant r)
\leqslant\frac{\mathbb{E}[|X|^n]}{r^n}.
\]
Vamos agora mostrar outra relação entre momentos e função característica:
quando o momento absoluto de ordem $n$ é finito, podemos derivar
$\varphi_X$ até a ordem $n$ e calcular suas derivadas por meio de esperanças.

\begin{theorem}[Derivadas da função característica]
\label{thm:derivadas-momentos}
\label{prop:man-derivadas-fc}
Seja $X$ uma variável aleatória com função característica $\varphi_X$.
Se $\mathbb{E}[|X|^n]<\infty$ para algum inteiro $n\geqslant1$, então, para
cada inteiro $k$ com $1\leqslant k\leqslant n$, a derivada $\varphi_X^{(k)}$
existe, é contínua em $\mathbb{R}$ e satisfaz
\[
\varphi_X^{(k)}(t)
=\mathbb{E}\bigl[(iX)^k e^{itX}\bigr],
\qquad t\in\mathbb{R}.
\]
\end{theorem}

\begin{proof}
Vamos derivar a esperança uma vez de cada vez, verificando em cada
passo a integrabilidade necessária. Na primeira derivada, explicitaremos
o argumento de convergência dominada com quocientes incrementais.
Nas demais, aplicaremos o \Cref{thm:derivacao-integral}.

Primeiro, a hipótese de momento absoluto de ordem $n$ garante os momentos
absolutos de ordens menores. Com efeito, se $1\leqslant j\leqslant n$, então
\[
|x|^j\leqslant1+|x|^n,\qquad x\in\mathbb{R}.
\]
Para $|x|\leqslant1$, temos $|x|^j\leqslant1$; para $|x|>1$, temos
$|x|^j\leqslant|x|^n$. Logo
\[
\mathbb{E}[|X|^j]\leqslant1+\mathbb{E}[|X|^n]<\infty,
\qquad 1\leqslant j\leqslant n.
\]

Comecemos pela primeira derivada. Fixe $t\in\mathbb{R}$.
Pela linearidade da esperança, para $h\neq0$,
\[
\frac{\varphi_X(t+h)-\varphi_X(t)}{h}
=\mathbb{E}\left[
e^{itX}\frac{e^{ihX}-1}{h}
\right].
\]
Para justificar a passagem ao limite, tomamos uma sequência
$h_m\to0$, com $h_m\neq0$, e definimos
\[
\psi_m\equiv e^{itX}\frac{e^{ih_mX}-1}{h_m}.
\]
Para cada $\omega\in\Omega$, temos
$\psi_m(\omega)\to iX(\omega)e^{itX(\omega)}$.
Além disso, a identidade
\[
e^{iu}-1=i\int_0^u e^{is}\,ds,
\qquad u\in\mathbb{R},
\]
fornece $|e^{iu}-1|\leqslant|u|$. Portanto,
\[
|\psi_m|
\leqslant\frac{|h_mX|}{|h_m|}
=|X|.
\]
Como $\mathbb{E}[|X|]<\infty$, a convergência dominada, aplicada
às partes real e imaginária, dá
\[
\frac{\varphi_X(t+h_m)-\varphi_X(t)}{h_m}
\longrightarrow\mathbb{E}[iX e^{itX}].
\]
Como a sequência $(h_m)$ foi arbitrária, concluímos que
$\varphi_X'(t)=\mathbb{E}[iX e^{itX}]$.

Suponha agora que, para algum $k<n$, já tenhamos obtido
\[
\varphi_X^{(k)}(t)=\mathbb{E}\bigl[(iX)^k e^{itX}\bigr].
\]
Para derivar mais uma vez, tomamos $g(t,x)=(ix)^k e^{itx}$. Então
\[
\frac{\partial g}{\partial t}(t,X)
=(iX)^{k+1}e^{itX},
\qquad
\left|\frac{\partial g}{\partial t}(t,X)\right|
=|X|^{k+1}.
\]
A variável $g(t,X)$ é integrável, pois seu módulo é $|X|^k$, e a derivada é
majorada pela variável integrável $|X|^{k+1}$. Aplicando novamente o
\Cref{thm:derivacao-integral}, concluímos que
\[
\varphi_X^{(k+1)}(t)
=\frac{d}{dt}\mathbb{E}\bigl[(iX)^k e^{itX}\bigr]
=\mathbb{E}\bigl[(iX)^{k+1}e^{itX}\bigr].
\]
A indução fornece, portanto, a fórmula para todas as ordens de $1$ a $n$.

Resta verificar a continuidade. Fixe $k\in\{1,\ldots,n\}$ e tome uma
sequência $t_m\to t$. Para cada resultado do experimento,
\[
(iX)^k e^{it_mX}\longrightarrow(iX)^k e^{itX},
\qquad
\bigl|(iX)^k e^{it_mX}\bigr|=|X|^k.
\]
Como $\mathbb{E}[|X|^k]<\infty$, a convergência dominada, aplicada às partes
real e imaginária, dá
\[
\varphi_X^{(k)}(t_m)
=\mathbb{E}\bigl[(iX)^k e^{it_mX}\bigr]
\longrightarrow
\mathbb{E}\bigl[(iX)^k e^{itX}\bigr]
=\varphi_X^{(k)}(t).
\]
Logo $\varphi_X^{(k)}$ é contínua para cada $k\leqslant n$.
\end{proof}

Se $F_X$ é a função de distribuição de $X$, a fórmula do teorema
também pode ser escrita como
\[
\varphi_X^{(k)}(t)
=\int_{\mathbb{R}}(ix)^k e^{itx}\,dF_X(x),
\qquad 1\leqslant k\leqslant n.
\]

Tomando $t=0$ no teorema e usando $e^0=1$, obtemos
\[
\varphi_X^{(k)}(0)=i^k\,\mathbb{E}[X^k],
\qquad 1\leqslant k\leqslant n.
\]
Em particular, se $\mathbb{E}[|X|]<\infty$, então
\[
\varphi_X'(0)=i\,\mathbb{E}[X].
\]
Se $\mathbb{E}[X^2]<\infty$, temos também
\[
\varphi_X''(0)=i^2\,\mathbb{E}[X^2]=-\mathbb{E}[X^2].
\]
Nesse caso, a média é $\mathbb{E}[X]=-i\varphi_X'(0)$, e a variância é
\[
\begin{aligned}
\operatorname{Var}(X)
&=\mathbb{E}[X^2]-\bigl(\mathbb{E}[X]\bigr)^2
\\[0.3cm]
&=-\varphi_X''(0)-\bigl(-i\varphi_X'(0)\bigr)^2
=-\varphi_X''(0)+\bigl(\varphi_X'(0)\bigr)^2.
\end{aligned}
\]

Por exemplo, no \Cref{ex:rademacher} vimos que uma variável $R$ que assume
$-1$ e $1$ com probabilidades iguais tem função característica
$\varphi_R(t)=\cos t$. Como $\varphi_R'(0)=0$ e
$\varphi_R''(0)=-1$, as fórmulas dão
\[
\mathbb{E}[R]=0,
\qquad
\mathbb{E}[R^2]=1,
\]
em acordo com o cálculo direto a partir da distribuição de $R$.

A existência das derivadas também permite aproximar $\varphi_X$ perto da origem.
Se $\mathbb{E}[|X|^n]<\infty$, aplicamos a fórmula de Taylor às partes real e
imaginária de $\varphi_X$ e substituímos as derivadas pelos momentos:
\[
\varphi_X(t)
=1+\sum_{k=1}^{n}\frac{(it)^k}{k!}\,\mathbb{E}[X^k]
+o(t^n),
\qquad t\to0.
\]
Aqui $o(t^n)$ representa um resto cujo quociente por $t^n$ tende a zero
quando $t\to0$. Em particular, se $\mathbb{E}[X^2]<\infty$,
\[
\varphi_X(t)
=1+it\,\mathbb{E}[X]
-\frac{t^2}{2}\,\mathbb{E}[X^2]+o(t^2).
\]
Para essa conclusão, não precisamos supor que o terceiro momento seja finito.

Na próxima subseção, obteremos uma cota explícita para o resto da expansão da
exponencial. Ela permitirá estimar o erro antes de tomar a esperança e será
útil no estudo de somas de variáveis aleatórias.

\subsection{Uma cota para o resto de Taylor}\label{subsec:resto-taylor}

O \Cref{thm:derivadas-momentos} já fornece uma expansão assintótica de
$\varphi_X$ na origem. Vamos agora obter uma cota explícita para o
resto, estimando primeiro o erro na aproximação de $e^{ix}$ por seu
polinômio de Taylor. Assim poderemos controlar esse erro antes de
tomar a esperança, como faremos na prova do Teorema do Limite Central.

\begin{lemma}[Cota para o resto de Taylor]\label{lem:resto-taylor}
\label{lem:man-resto-taylor}
Para todo inteiro $n\geqslant0$ e todo $x\in\mathbb{R}$,
\[
\left|e^{ix}-\sum_{m=0}^n\frac{(ix)^m}{m!}\right|
\leqslant
\min\left\{
\frac{|x|^{n+1}}{(n+1)!},\,
\frac{2|x|^n}{n!}
\right\}.
\]
Quando $n=0$, a segunda cota é entendida como $2$, de modo que
a desigualdade se reduz a $|e^{ix}-1|\leqslant\min\{|x|,2\}$.
\end{lemma}

As duas cotas cumprem funções diferentes. A primeira descreve o
comportamento do resto para $|x|$ pequeno, pois contém a potência
$|x|^{n+1}$. A segunda cresce apenas como $|x|^n$ e será útil para
valores grandes de $|x|$. Quando substituirmos $x$ por $tX$, essa
diferença permitirá trabalhar sob uma hipótese de momento de
ordem $n$, sem exigir o momento de ordem $n+1$.

\begin{proof}
Vamos obter uma identidade integral para o resto e estimá-la de duas
maneiras.

\smallskip
\noindent\textbf{Primeira etapa: a identidade do resto.}
Integrando por partes com
\[
u=e^{is},
\qquad
dv=(x-s)^n\,ds,
\qquad
du=ie^{is}\,ds,
\qquad
v=-\frac{(x-s)^{n+1}}{n+1},
\]
obtemos
\begin{equation}\label{eq:man-identidade-partes}
\begin{aligned}
\int_0^x(x-s)^n e^{is}\,ds
&=\left[-\frac{(x-s)^{n+1}}{n+1}e^{is}\right]_{s=0}^{s=x}
+\frac{i}{n+1}\int_0^x(x-s)^{n+1}e^{is}\,ds
\\[0.3cm]
&=\frac{x^{n+1}}{n+1}
+\frac{i}{n+1}\int_0^x(x-s)^{n+1}e^{is}\,ds.
\end{aligned}
\end{equation}
O termo de fronteira se anula em $s=x$ e vale
$-x^{n+1}/(n+1)$ em $s=0$. A identidade também vale para $x<0$,
pois as integrais são orientadas.

O ponto de partida é
\[
e^{ix}-1=i\int_0^x e^{is}\,ds.
\]
Usando \eqref{eq:man-identidade-partes} com $n=0$, encontramos
\[
e^{ix}=1+ix+i^2\int_0^x(x-s)e^{is}\,ds.
\]
Usando-a mais uma vez, agora com $n=1$, obtemos
\[
e^{ix}
=1+ix+\frac{(ix)^2}{2}
+\frac{i^3}{2}\int_0^x(x-s)^2e^{is}\,ds.
\]
Esses cálculos sugerem a identidade
\begin{equation}\label{eq:man-resto-integral}
e^{ix}-\sum_{m=0}^n\frac{(ix)^m}{m!}
=\frac{i^{n+1}}{n!}\int_0^x(x-s)^n e^{is}\,ds.
\end{equation}
Vamos verificá-la por indução. O caso $n=0$ é a identidade inicial.
Suponha que ela seja válida para certo $n\geqslant0$. Pela
\eqref{eq:man-identidade-partes}, seu lado direito é
\[
\frac{i^{n+1}x^{n+1}}{(n+1)!}
+\frac{i^{n+2}}{(n+1)!}
\int_0^x(x-s)^{n+1}e^{is}\,ds.
\]
Passando a primeira parcela para o lado esquerdo, obtemos
\eqref{eq:man-resto-integral} com $n$ substituído por $n+1$.
Isso conclui a indução.

\smallskip
\noindent\textbf{Segunda etapa: a cota de ordem $|x|^{n+1}$.}
Como $|e^{is}|=1$, a identidade do resto dá
\[
\left|
\frac{i^{n+1}}{n!}\int_0^x(x-s)^n e^{is}\,ds
\right|
\leqslant
\frac{1}{n!}\int_0^{|x|}(|x|-u)^n\,du
=\frac{|x|^{n+1}}{(n+1)!}.
\]
Para $x<0$, invertemos os extremos da integral e fazemos $u=-s$;
para $x\geqslant0$, basta tomar $u=s$. Assim, a mesma estimativa
cobre os dois sinais de $x$.

\smallskip
\noindent\textbf{Terceira etapa: a cota de ordem $|x|^n$.}
Para diminuir em uma unidade a potência de $|x|$, vamos reduzir
o grau do núcleo $(x-s)^n$ por integração por partes.
Suponha $n\geqslant1$ e tome
\[
u=(x-s)^n,
\qquad
dv=e^{is}\,ds,
\qquad
du=-n(x-s)^{n-1}\,ds,
\qquad
v=\frac{e^{is}}{i}.
\]
Então
\begin{align*}
\int_0^x(x-s)^n e^{is}\,ds
&=\left[\frac{(x-s)^n e^{is}}{i}\right]_{s=0}^{s=x}
+\frac{n}{i}\int_0^x(x-s)^{n-1}e^{is}\,ds
\\[0.3cm]
&=-\frac{x^n}{i}
+\frac{n}{i}\int_0^x(x-s)^{n-1}e^{is}\,ds.
\end{align*}
Multiplicando por $i/n$ e escrevendo o termo de fronteira
como uma integral,
\[
\frac{x^n}{n}=\int_0^x(x-s)^{n-1}\,ds,
\]
podemos reunir as duas parcelas:
\[
\frac{i}{n}\int_0^x(x-s)^n e^{is}\,ds
=\int_0^x(x-s)^{n-1}(e^{is}-1)\,ds.
\]
Portanto,
\[
\frac{i^{n+1}}{n!}\int_0^x(x-s)^n e^{is}\,ds
=\frac{i^n}{(n-1)!}
\int_0^x(x-s)^{n-1}(e^{is}-1)\,ds.
\]
Agora usamos $|e^{is}-1|\leqslant2$. Como na segunda etapa,
consideramos o sinal de $x$ para obter
\[
\left|
\frac{i^{n+1}}{n!}\int_0^x(x-s)^n e^{is}\,ds
\right|
\leqslant
\frac{2}{(n-1)!}\int_0^{|x|}(|x|-u)^{n-1}\,du
=\frac{2|x|^n}{n!}.
\]
A redução do grau do núcleo é o que permite substituir o crescimento
$|x|^{n+1}$ por $|x|^n$.

Para $n=0$, a terceira etapa não se aplica, pois divide por $n$.
Nesse caso, a segunda cota segue diretamente de
$|e^{ix}-1|\leqslant|e^{ix}|+1=2$.
Como as duas estimativas valem simultaneamente, podemos tomar
o mínimo entre elas.
\end{proof}

Substituindo $x$ por $tx$, obtemos a forma que será aplicada
às variáveis aleatórias.

\begin{corollary}\label{cor:man-resto-taylor}
Para quaisquer $t,x\in\mathbb{R}$ e todo inteiro $n\geqslant0$,
\[
\left|e^{itx}-\sum_{m=0}^n\frac{(itx)^m}{m!}\right|
\leqslant
\min\left\{
\frac{|tx|^{n+1}}{(n+1)!},\,
\frac{2|tx|^n}{n!}
\right\},
\]
com a mesma convenção do \Cref{lem:resto-taylor} quando $n=0$.
\end{corollary}

Se $n\geqslant1$ e $\mathbb{E}[|X|^n]<\infty$, todos os termos
do polinômio são integráveis, pelo argumento usado na prova do
\Cref{thm:derivadas-momentos}. O mínimo também é integrável,
pois é majorado por $2|t|^n|X|^n/n!$. Pela linearidade da
esperança e pela desigualdade $|\mathbb{E}[Y]|\leqslant\mathbb{E}[|Y|]$,
válida para variáveis aleatórias complexas integráveis,
\begin{equation}\label{eq:cota-resto-taylor}
\begin{aligned}
\left|
\varphi_X(t)-\sum_{m=0}^n\frac{(it)^m}{m!}\mathbb{E}[X^m]
\right|
&\leqslant
\mathbb{E}\left[
\left|e^{itX}-\sum_{m=0}^n\frac{(itX)^m}{m!}\right|
\right]
\\[0.3cm]
&\leqslant
\mathbb{E}\left[
\min\left\{
\frac{|tX|^{n+1}}{(n+1)!},\,
\frac{2|tX|^n}{n!}
\right\}
\right].
\end{aligned}
\end{equation}
Essa cota vale para todo $t\in\mathbb{R}$.
Se $\mathbb{E}[|X|^{n+1}]<\infty$, a primeira parcela do mínimo
fornece uma cota de ordem $|t|^{n+1}$ para o resto.
Se dispomos apenas de $\mathbb{E}[|X|^n]<\infty$, a segunda
fornece uma cota de ordem $|t|^n$. Na próxima subseção, veremos
como usar simultaneamente as duas parcelas para obter um resto
$o(t^2)$ sob a hipótese de segundo momento finito.

\subsection{A expansão de segunda ordem e sua recíproca}\label{subsec:segunda-ordem}

Vamos aplicar \eqref{eq:cota-resto-taylor} com $n=2$.
Se $\mathbb{E}[X^2]<\infty$, então $\mathbb{E}[|X|]<\infty$,
e os coeficientes do polinômio
\[
1+it\,\mathbb{E}[X]
+\frac{(it)^2}{2}\mathbb{E}[X^2]
=1+it\,\mathbb{E}[X]-\frac{t^2}{2}\mathbb{E}[X^2]
\]
são finitos. O objetivo é mostrar que, após a divisão por $t^2$,
o resto tende a zero, mesmo quando o terceiro momento absoluto
não é finito.

\begin{theorem}[Expansão de segunda ordem]\label{thm:taylor-segunda-ordem}
\label{thm:man-expansao-segunda-ordem}
Seja $X$ uma variável aleatória com função característica $\varphi_X$.
Se $\mathbb{E}[X^2]<\infty$, então
\[
\varphi_X(t)
=1+it\,\mathbb{E}[X]-\frac{t^2}{2}\mathbb{E}[X^2]+g(t),
\]
onde $g(t)=o(t^2)$ quando $t\to0$, isto é,
$g(t)/t^2\to0$.
\end{theorem}

\begin{proof}
Pela linearidade da esperança, podemos escrever
\begin{align*}
\varphi_X(t)
&=\mathbb{E}\left[
\left(e^{itX}-1-itX+\frac{t^2}{2}X^2\right)
+1+itX-\frac{t^2}{2}X^2
\right]
\\[0.3cm]
&=1+it\,\mathbb{E}[X]-\frac{t^2}{2}\mathbb{E}[X^2]+g(t),
\end{align*}
onde
\[
g(t)\equiv
\mathbb{E}\left[e^{itX}-1-itX+\frac{t^2}{2}X^2\right].
\]
A integrabilidade de todas as parcelas segue da hipótese de segundo
momento finito. Em particular, $g(0)=0$.

Pela \eqref{eq:cota-resto-taylor}, com $n=2$, temos
\[
|g(t)|
\leqslant
\mathbb{E}\left[
\min\left\{
\frac{|tX|^3}{6},\,t^2|X|^2
\right\}
\right].
\]
Para $t\neq0$, isolamos o fator $t^2$ dentro da esperança e obtemos
\begin{equation}\label{eq:resto-dividido}
\frac{|g(t)|}{t^2}
\leqslant
\mathbb{E}\left[
\min\left\{
\frac{|t|\,|X|^3}{6},\,|X|^2
\right\}
\right].
\end{equation}
Basta mostrar que o lado direito tende a zero.
Defina
\[
Y_t\equiv
\min\left\{
\frac{|t|\,|X|^3}{6},\,|X|^2
\right\}.
\]
Para cada $\omega\in\Omega$, o primeiro argumento do mínimo tende
a zero quando $t\to0$, logo $Y_t(\omega)\to0$.
Além disso,
\[
0\leqslant Y_t\leqslant|X|^2,
\]
e a variável dominante é integrável por hipótese.
Pelo Teorema da Convergência Dominada, aplicado ao longo de qualquer
sequência $t_m\to0$, temos $\mathbb{E}[Y_{t_m}]\to0$.
Consequentemente, $\mathbb{E}[Y_t]\to0$ quando $t\to0$.
A desigualdade \eqref{eq:resto-dividido} dá então
$|g(t)|/t^2\to0$, como queríamos.
\end{proof}

\begin{remark}[Por que a cota com o mínimo]\label{rem:cota-minimo}
Se usássemos apenas a primeira cota do \Cref{lem:resto-taylor},
teríamos
\[
|g(t)|\leqslant\frac{|t|^3}{6}\mathbb{E}[|X|^3].
\]
Essa estimativa só forneceria o limite desejado sob a hipótese
adicional de terceiro momento absoluto finito.

O mínimo permite separar os papéis das duas cotas. Para
$|tX|\leqslant6$, temos $|tX|^3/6\leqslant t^2|X|^2$, e usamos
a cota cúbica. Para $|tX|>6$, usamos a cota quadrática.
Após a divisão por $t^2$, a primeira garante a convergência pontual
a zero, enquanto a segunda fornece a dominante integrável $|X|^2$.
Assim, a convergência dominada exige apenas o segundo momento.
\end{remark}

Vamos agora examinar uma recíproca. Pela propriedade
$\varphi_X(-h)=\overline{\varphi_X(h)}$ e pela igualdade
$\varphi_X(0)=1$, o quociente
\[
D_X(h)\equiv
\frac{\varphi_X(h)-2\varphi_X(0)+\varphi_X(-h)}{h^2}
=\frac{2\bigl(\operatorname{Re}\varphi_X(h)-1\bigr)}{h^2},
\qquad h\neq0,
\]
é real e não positivo. A expansão de segunda ordem mostra que,
se $\mathbb{E}[X^2]<\infty$, então $D_X(h)\to-\mathbb{E}[X^2]$.
O próximo resultado recupera a finitude do segundo momento sob
uma hipótese mais fraca sobre esse quociente.

\begin{theorem}[Segundo momento e diferença segunda]\label{thm:segundo-momento-finito}
Seja $X$ uma variável aleatória com função característica $\varphi_X$.
Se
\[
\limsup_{h\downarrow0}
\frac{\varphi_X(h)-2\varphi_X(0)+\varphi_X(-h)}{h^2}
>-\infty,
\]
então $\mathbb{E}[X^2]<\infty$.
\end{theorem}

\begin{proof}
Seja $F_X$ a função de distribuição de $X$. Pela linearidade
da integral, aplicada aos integrandos limitados,
\[
D_X(h)
=\int_{\mathbb{R}}
\frac{e^{ihx}-2+e^{-ihx}}{h^2}\,dF_X(x).
\]
Como $e^{ihx}+e^{-ihx}=2\cos(hx)$,
\begin{equation}\label{eq:integrando-segunda-diferenca}
\frac{e^{ihx}-2+e^{-ihx}}{h^2}
=-\frac{2\bigl(1-\cos(hx)\bigr)}{h^2}
\leqslant0.
\end{equation}
Para aplicar o lema de Fatou, tomamos o oposto desse integrando:
\[
q_h(x)\equiv
\frac{2\bigl(1-\cos(hx)\bigr)}{h^2}\geqslant0.
\]
O limite $(1-\cos u)/u^2\to1/2$, quando $u\to0$, mostra que
$q_h(x)\to x^2$ para cada $x\in\mathbb{R}$.
Para $x=0$, essa afirmação é imediata; para $x\neq0$,
basta escrever $u=hx$.

Ponha $L\equiv\limsup_{h\downarrow0}D_X(h)$. Pela hipótese
e pelo fato de $D_X(h)\leqslant0$, temos $-\infty<L\leqslant0$.
Escolha uma sequência $h_m\downarrow0$ tal que
$D_X(h_m)\to L$. Aplicando o lema de Fatou às funções
não negativas $q_{h_m}$, obtemos
\begin{align*}
\mathbb{E}[X^2]
&=\int_{\mathbb{R}}x^2\,dF_X(x)
=\int_{\mathbb{R}}\liminf_{m\to\infty}q_{h_m}(x)\,dF_X(x)
\\[0.3cm]
&\leqslant
\liminf_{m\to\infty}\int_{\mathbb{R}}q_{h_m}(x)\,dF_X(x)
\\[0.3cm]
&=\liminf_{m\to\infty}\bigl(-D_X(h_m)\bigr)
=-L<\infty.
\end{align*}
Isso prova a finitude do segundo momento.
\end{proof}

Combinando os dois teoremas, a finitude do segundo momento
é equivalente à existência de um limite finito para a diferença
segunda simétrica $D_X(h)$ na origem. Nesse caso, o limite é
$-\mathbb{E}[X^2]$. O \Cref{thm:segundo-momento-finito} mostra
que basta supor que o limite superior não seja $-\infty$.

A diferença segunda fornece um critério sobre valores de
$\varphi_X$ perto da origem, sem supor previamente a existência
de derivadas. A derivada usual de $\varphi_X$, como função de
variável real com valores complexos, é definida pelos quocientes
incrementais usuais. Se $\varphi_X''(0)$ existe, a fórmula de Taylor
dá $D_X(h)\to\varphi_X''(0)$, logo o segundo momento é finito.
Reciprocamente, o \Cref{thm:derivadas-momentos} garante que o
segundo momento finito implica a existência dessa derivada, com
$\varphi_X''(0)=-\mathbb{E}[X^2]$.


\section{Teorema do Limite Central}\label{sec:tlc}

Vamos combinar a expansão de segunda ordem da função característica com o
limite exponencial da \Cref{sec:preliminares-complexos}. A independência
produzirá o produto, e a normalização por $\sqrt{n}$ fará aparecer um
termo de ordem $1/n$ em cada fator.

\begin{theorem}[Teorema do Limite Central, caso i.i.d.]\label{thm:man-tlc-iid}
Seja $(X_n)_{n\geqslant1}$ uma sequência de variáveis aleatórias
independentes e identicamente distribuídas, com média
$\mu=\mathbb{E}[X_1]$ e variância
$\sigma^2=\operatorname{Var}(X_1)\in(0,\infty)$.
Então
\[
Z_n\equiv\frac{1}{\sigma\sqrt{n}}\sum_{j=1}^n(X_j-\mu)
\Rightarrow Z,
\]
onde $Z$ tem distribuição normal padrão.
\end{theorem}

\begin{proof}
Primeiro, vamos reduzir a demonstração ao caso de média zero e
variância um. As variáveis
\[
Y_j\equiv\frac{X_j-\mu}{\sigma},
\qquad j\geqslant1,
\]
continuam independentes e identicamente distribuídas e satisfazem
\[
\mathbb{E}[Y_j]=0,
\qquad
\operatorname{Var}(Y_j)=1.
\]
Além disso,
\[
\frac{1}{\sqrt{n}}\sum_{j=1}^nY_j
=\frac{1}{\sigma\sqrt{n}}\sum_{j=1}^n(X_j-\mu).
\]
Portanto, basta provar o resultado quando
$\mathbb{E}[X_1]=0$ e $\operatorname{Var}(X_1)=1$.
Assumiremos essas condições no restante da prova. Nesse caso,
$\mathbb{E}[X_1^2]=1$ e
\[
Z_n=\frac{1}{\sqrt{n}}\sum_{j=1}^nX_j.
\]

Pelo \Cref{ex:normal}, a função característica da normal padrão
é $e^{-t^2/2}$. Vamos mostrar que
\[
\varphi_{Z_n}(t)\longrightarrow e^{-t^2/2},
\qquad t\in\mathbb{R},
\]
e então aplicar o Teorema da Continuidade de Lévy.
Para $t=0$, a igualdade $\varphi_{Z_n}(0)=1$ já fornece
o limite. Fixe agora $t\neq0$.

Pela independência e pelo fato de as parcelas terem a mesma
distribuição, obtemos
\begin{align*}
\varphi_{Z_n}(t)
&=\mathbb{E}\left[
e^{i(t/\sqrt{n})\sum_{j=1}^nX_j}
\right]
\\[0.3cm]
&=\mathbb{E}\left[
\prod_{j=1}^n e^{i(t/\sqrt{n})X_j}
\right]
\\[0.3cm]
&=\prod_{j=1}^n\mathbb{E}\left[e^{i(t/\sqrt{n})X_j}\right]
\\[0.3cm]
&=\prod_{j=1}^n\mathbb{E}\left[e^{i(t/\sqrt{n})X_1}\right]
=\left(\varphi_{X_1}\left(\frac{t}{\sqrt{n}}\right)\right)^n.
\end{align*}
A terceira linha usa a independência das variáveis
$e^{i(t/\sqrt{n})X_j}$, que são integráveis por terem módulo $1$.
A quarta usa a igualdade das distribuições. Esse é o mecanismo
da regra do produto recordada no \Cref{thm:fc-soma-independentes}.

Pelo \Cref{thm:taylor-segunda-ordem}, existe uma função $g$ tal que
\[
\varphi_{X_1}(u)
=1-\frac{u^2}{2}+g(u),
\qquad g(u)=o(u^2),
\]
pois $\mathbb{E}[X_1]=0$ e $\mathbb{E}[X_1^2]=1$.
Logo,
\begin{equation}\label{eq:man-fatoracao-zn}
\begin{aligned}
\varphi_{Z_n}(t)
&=\left(
1-\frac{t^2}{2n}+g\left(\frac{t}{\sqrt{n}}\right)
\right)^n
\\[0.3cm]
&=\left(1+\frac{c_n(t)}{n}\right)^n,
\end{aligned}
\end{equation}
onde
\[
c_n(t)\equiv-\frac{t^2}{2}
+n\,g\left(\frac{t}{\sqrt{n}}\right).
\]
Queremos mostrar que $c_n(t)\to-t^2/2$. Para isso, precisamos
controlar o erro depois da multiplicação por $n$.

A estimativa \eqref{eq:resto-dividido}, aplicada à variável $X_1$,
fornece
\[
\left|g\left(\frac{t}{\sqrt{n}}\right)\right|
\leqslant
\frac{t^2}{n}\,
\mathbb{E}\left[
\min\left\{
\frac{|t|\,|X_1|^3}{6\sqrt{n}},\,|X_1|^2
\right\}
\right].
\]
Multiplicando por $n$, obtemos
\[
\left|n\,g\left(\frac{t}{\sqrt{n}}\right)\right|
\leqslant
t^2\,\mathbb{E}\left[
\min\left\{
\frac{|t|\,|X_1|^3}{6\sqrt{n}},\,|X_1|^2
\right\}
\right]
\longrightarrow0.
\]
De fato, o integrando converge pontualmente a zero e é majorado
por $|X_1|^2$, que é integrável. O limite segue do Teorema da
Convergência Dominada.

Concluímos que $c_n(t)\to-t^2/2$. Pelo
\Cref{thm:man-limite-exp} e por \eqref{eq:man-fatoracao-zn},
\[
\varphi_{Z_n}(t)
=\left(1+\frac{c_n(t)}{n}\right)^n
\longrightarrow e^{-t^2/2}.
\]
A função limite é a função característica da normal padrão e
é contínua na origem. O \Cref{thm:continuidade} dá
$Z_n\Rightarrow Z$, como queríamos.
\end{proof}

Vamos justificar a formulação por intervalos apresentada na
introdução. Denote por $F_n$ a função de distribuição de $Z_n$
e por $\Phi$ a da normal padrão. Como $\Phi$ é contínua em toda
a reta, o teorema garante que $F_n(x)\to\Phi(x)$ para todo
$x\in\mathbb{R}$. Para $a<b$ e $\delta>0$,
\[
F_n(b)-F_n(a)
\leqslant\mathbb{P}(a\leqslant Z_n\leqslant b)
\leqslant F_n(b)-F_n(a-\delta).
\]
Passando aos limites inferior e superior em $n$ e depois fazendo
$\delta\downarrow0$, a continuidade de $\Phi$ fornece
\[
\mathbb{P}(a\leqslant Z_n\leqslant b)
\longrightarrow
\Phi(b)-\Phi(a)
=\frac{1}{\sqrt{2\pi}}\int_a^b e^{-x^2/2}\,dx.
\]
Isso mostra também que a inclusão dos extremos do intervalo
não altera o limite.

\section{Exercícios Propostos}\label{sec:exercicios}

Os exercícios retomam a relação entre momentos e funções características,
o controle das caudas e o comportamento de segunda ordem na origem.

\begin{exercise}[Momentos da normal padrão]
\label{exercise:momentos-normal}
Seja $X$ uma variável aleatória com distribuição normal padrão.
Verifique primeiro, a partir da densidade, que
$\mathbb{E}[|X|^m]<\infty$ para todo inteiro $m\geqslant1$.
Use então o \Cref{thm:derivadas-momentos} e o desenvolvimento em série de potências de
$e^{-t^{2}/2}$ para mostrar que, para todo $n\geqslant1$,
\[
  \mathbb{E}\bigl[X^{2n}\bigr]=\frac{(2n)!}{2^{n}\,n!}
  =(2n-1)(2n-3)\cdots3\cdot1\equiv(2n-1)!!,
\]
onde $(2n-1)!!$ é, por definição, o produto dos inteiros positivos ímpares que
não excedem $2n-1$, com a convenção $(-1)!!=1$. Em particular,
$\mathbb{E}[X^{2}]=1$ e $\mathbb{E}[X^{4}]=3$. Mostre também que os momentos de
ordem ímpar se anulam, por simetria da distribuição:
$\mathbb{E}[X^{2n-1}]=0$ para $n\geqslant1$.

\emph{Sugestão:} para a integrabilidade, observe que, para cada
$m\geqslant1$, existe $C_m<\infty$ tal que
$|x|^m e^{-x^2/2}\leqslant C_m e^{-x^2/4}$ para todo $x\in\mathbb{R}$.
A função característica da normal padrão é
$\varphi_X(t)=e^{-t^{2}/2}=\sum_{n\geqslant0}\frac{(-1)^{n}}{2^{n}n!}t^{2n}$, e
\Cref{thm:derivadas-momentos} dá
$\varphi_X^{(2n)}(0)=i^{2n}\mathbb{E}[X^{2n}]=(-1)^{n}\mathbb{E}[X^{2n}]$;
compare os coeficientes das duas expressões de $\varphi_X$. Para os momentos
ímpares, use $X\overset{d}{=}-X$.
\end{exercise}

\begin{exercise}[Equicontinuidade das funções características]\label{exercise:equicontinuidade}
Dizemos que uma família de variáveis aleatórias $\{X_i:i\in I\}$ é
\emph{condensada} se, para todo $\varepsilon>0$, existe $M>0$ tal que
\[
\mathbb{P}(|X_i|>M)<\varepsilon
\qquad\text{para todo }i\in I.
\]
Denotando por $\varphi_{X_i}$ a função característica de $X_i$, dizemos que a
família $\{\varphi_{X_i}:i\in I\}$ é \emph{equicontínua} se, para todo
$\varepsilon>0$, existe $\delta>0$ tal que $|h|<\delta$ implica
$|\varphi_{X_i}(t+h)-\varphi_{X_i}(t)|<\varepsilon$ para todos $i\in I$ e
$t\in\mathbb{R}$.

\begin{enumerate}
\item[(i)] Suponha que $\{X_i:i\in I\}$ seja condensada e mostre que
$\{\varphi_{X_i}:i\in I\}$ é equicontínua.

\emph{Sugestão:} pelo item~(d) do \Cref{thm:propriedades-fc},
\[
|\varphi_{X_i}(t+h)-\varphi_{X_i}(t)|
\leqslant\mathbb{E}\bigl[|e^{ihX_i}-1|\bigr].
\]
Use $|e^{i\theta}-1|\leqslant\min\{|\theta|,2\}$, obtida no
\Cref{lem:resto-taylor} com $n=0$, para mostrar que, para todo $M>0$,
\[
|\varphi_{X_i}(t+h)-\varphi_{X_i}(t)|
\leqslant |h|M+2\mathbb{P}(|X_i|>M).
\]
Escolha primeiro $M$ e depois $\delta$ em função de $\varepsilon$.

\item[(ii)] Suponha que $X_n\Rightarrow X_\infty$ e denote por
$\varphi_{X_\infty}$ a função característica de $X_\infty$. Mostre que a
família $\{X_n:n\in\mathbb{N}\}\cup\{X_\infty\}$ é condensada e conclua que
$\varphi_{X_n}\to\varphi_{X_\infty}$ uniformemente sobre cada intervalo limitado.

\emph{Sugestão:} para obter a condensação, escolha $M$ de modo que $-M$ e $M$
sejam pontos de continuidade de $F_{X_\infty}$ e que
$\mathbb{P}(|X_\infty|\geqslant M)$ seja pequena. Use a convergência das
funções de distribuição e, depois, aumente $M$ para incluir os finitos
índices iniciais. Aplique o item~(i) e o \Cref{thm:continuidade}. Dado um
intervalo limitado, cubra-o por um número finito de intervalos de raio
$\delta$, centrados em $t_1,\ldots,t_k$, e escreva
\[
\begin{aligned}
|\varphi_{X_n}(t)-\varphi_{X_\infty}(t)|
&\leqslant|\varphi_{X_n}(t)-\varphi_{X_n}(t_j)|
+|\varphi_{X_n}(t_j)-\varphi_{X_\infty}(t_j)|\\
&\quad+|\varphi_{X_\infty}(t_j)-\varphi_{X_\infty}(t)|.
\end{aligned}
\]
Use a equicontinuidade na primeira e na terceira parcelas e a convergência
pontual em $t_j$ na parcela do meio.

\item[(iii)] Dê um exemplo em que $X_n\Rightarrow X_\infty$, mas
$\varphi_{X_n}$ não converge para $\varphi_{X_\infty}$ uniformemente em
$\mathbb{R}$.

\emph{Sugestão:} tome $X_n\equiv1/n$ e $X_\infty\equiv0$ e observe que
$\sup_{t\in\mathbb{R}}|e^{it/n}-1|=2$.
\end{enumerate}
\end{exercise}

\begin{exercise}[Comportamento de segunda ordem na origem]\label{exercise:segunda-derivada-nula}
Seja $X$ uma variável aleatória com função de distribuição $F_X$ e função
característica $\varphi_X$. Suponha que
\[
  \lim_{t\downarrow0}\frac{\varphi_X(t)-1}{t^2}=c,
  \qquad c\in\mathbb{R}.
\]
Mostre que $\mathbb{E}[X]=0$ e que $\mathbb{E}[X^2]=-2c<\infty$. Em
particular, se $\varphi_X(t)=1+o(t^2)$ quando $t\to0$, então $\varphi_X(t)\equiv1$
e $X=0$ quase certamente.

\emph{Sugestão:} da hipótese segue que
$\bigl(1-\operatorname{Re}\varphi_X(t)\bigr)/t^2\leqslant C$ para
$t\in(0,\delta]$. Ora,
\[
\frac{1-\operatorname{Re}\varphi_X(t)}{t^2}
=\int_{\mathbb{R}}\frac{1-\cos(tx)}{t^2}\,dF_X(x),
\]
e o integrando é não negativo e converge para $x^2/2$ quando $t\downarrow0$,
para cada $x$ fixado. Aplicando o lema de Fatou ao longo de uma sequência
$t_k\downarrow0$, obtemos
$\mathbb{E}[X^2]\leqslant2C<\infty$. Com o segundo momento finito, a expansão
de segunda ordem de \Cref{thm:taylor-segunda-ordem} dá
$\varphi_X(t)=1+it\,\mathbb{E}[X]-\frac{t^2}{2}\mathbb{E}[X^2]+o(t^2)$; divida
por $t^2$ e faça $t\downarrow0$ para concluir que $\mathbb{E}[X]=0$ e que
$c=-\frac{1}{2}\mathbb{E}[X^2]$.
\end{exercise}
