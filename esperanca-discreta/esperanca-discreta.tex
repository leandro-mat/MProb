% --- RESUMO (ABSTRACT) ---
\begin{abstract}
Estas notas estudam dois temas centrais da teoria das variáveis aleatórias
discretas: as funções de variáveis aleatórias e a esperança. Mostramos como a
função de probabilidade de \(Y = g(X)\) pode ser calculada a partir da função
de probabilidade de \(X\), com exemplos concretos como o caso em que \(g\) é
afim ou \(g(x) = x^2\). Em seguida, definimos a esperança de uma variável
aleatória discreta, discutimos sua interpretação como valor médio, média ponderada e como centro
de gravidade, e demonstramos a Fórmula da Mudança de Variáveis, que permite
calcular \(\mathbb{E}(g(X))\) diretamente, bem como as propriedades básicas da
esperança. Introduzimos também a variância como medida de dispersão em torno da
esperança e obtemos a fórmula \(\operatorname{Var}(X) =
\mathbb{E}(X^2) - \mathbb{E}(X)^2\). O texto inclui exemplos envolvendo as
distribuições de Poisson e geométrica e exercícios que complementam a
exposição. O texto segue de perto as Seções 2.3--2.4 do Capítulo 2 de Grimmett
e Welsh \cite{GrimmettWelsh2014}.
\end{abstract}

Estas notas dão continuidade ao desenvolvimento da teoria da probabilidade
iniciado nas notas \emph{Espaços de Probabilidade} e \emph{Variáveis
Aleatórias Discretas}. Nas primeiras foram introduzidos os ingredientes básicos
do modelo probabilístico: o espaço amostral, a \(\sigma\)-álgebra de eventos e
a medida de probabilidade; nas segundas, o conceito de variável aleatória
discreta e o de função de probabilidade. Aqui estudamos, primeiro, como
descrever variáveis aleatórias da forma \(Y = g(X)\), obtidas aplicando-se uma
função a uma variável aleatória discreta \(X\), e, depois, a noção de
\emph{esperança} de \(X\), que é, sem dúvida, um dos conceitos mais importantes
de toda a teoria das probabilidades, junto com o de \emph{variância}, que mede
o grau de dispersão de \(X\) em torno de sua esperança. Seguimos de perto a
exposição de \cite{GrimmettWelsh2014}, que recomendamos como leitura
complementar.


\section{Funções de Variáveis Aleatórias Discretas}
\label{sec:funcoes-de-variaveis}

Seja \(X\) uma variável aleatória discreta definida sobre um espaço de probabilidade
\((\Omega, \mathcal{F}, \mathbb{P})\) e seja \(g : \mathbb{R} \to \mathbb{R}\)
uma função arbitrária. A composição
\[
\omega\longmapsto Y(\omega) \equiv  g(X(\omega)) \qquad \forall \omega \in \Omega
\]
toma valores em \(g(\operatorname{Im} X)\), que é enumerável, e satisfaz
\(\{Y \in B\} = \{X \in g^{-1}(B)\} \in \mathcal{F}\) para todo boreliano
\(B \subseteq \mathbb{R}\); portanto \(Y \equiv g\circ X\) define uma variável
aleatória discreta sobre \((\Omega, \mathcal{F}, \mathbb{P})\).
Veja a Figura~\ref{fig:composicao-y-gx}.

São exemplos simples de funções de uma variável aleatória:
\[
\begin{aligned}
\text{se } g(x) = ax + b, &\quad \text{então } g(X) = aX + b,\\[0.3cm]
\text{se } g(x) = cx^2,  &\quad \text{então } g(X) = cX^2.
\end{aligned}
\]

A proposição abaixo, embora seja um resultado bastante simples, vai acabar se revelando uma poderosa
ferramenta quando estivermos interessados em calcular o valor médio ou a esperança de uma função de uma
variável aleatória. Observe que ela fornece uma maneira de representar a função de probabilidade 
de uma função de uma variável aleatória em termos da função de probabilidade da variável aleatória original.

\begin{proposition}
Seja $X$ uma variável aleatória discreta sobre um espaço de probabilidade $(\Omega,\mathcal{F},\mathbb{P})$ e
\(Y = g(X)\), onde $g:\mathbb{R}\to\mathbb{R}$ é uma função arbitrária. Então 
a função de probabilidade de \(Y\) é dada por
\begin{equation}\label{eq:funcao-massa-g}
p_Y(y) = \sum_{x \in \operatorname{Im} X \cap g^{-1}(y)} \mathbb{P}(X = x).
\end{equation}
\end{proposition}

\begin{proof}
Primeiro lembramos que se $A\subseteq \mathbb{R}$ é um conjunto arbitrário (enumerável ou não-enumerável), 
então o evento $\{X\in A\}$ pode ser decomposto da seguinte maneira como uma união enumerável de eventos
\begin{align*}
\{X\in A\} &= \{X\in A\} \cap \Omega 
\\[0.3cm]
&= \{X\in A\} \cap \bigcup_{x\in \operatorname{Im} X} \{X=x\}
\\[0.3cm]
&=
\bigcup_{x\in \operatorname{Im} X} \{X\in A\} \cap  \{X=x\}
\\[0.3cm]
&= \bigcup_{x\in \operatorname{Im} X\cap A} \{X=x\}.
\end{align*}

Fixe $y\in \mathbb{R}$ e tome $A=g^{-1}(y)$ na identidade acima. Como os eventos
$\{X=x\}$, $x\in \operatorname{Im} X$, são dois a dois disjuntos, temos da definição
de função de probabilidade de $Y$ e da $\sigma$-aditividade de $\mathbb{P}$ que
\begin{equation*}
\begin{aligned}
p_Y(y) = \mathbb{P}(Y = y) 
&= \mathbb{P}(g(X) = y) \\[0.3cm]
&= \mathbb{P}(X \in g^{-1}(y)) \\[0.3cm]
&= \sum_{x\in \operatorname{Im} X\cap g^{-1}(y)} \mathbb{P}(X = x).
\end{aligned}
\end{equation*}
Isso encerra a prova da proposição.
\end{proof}

Em alguns casos mais simples é possível exibir explicitamente os pontos da imagem inversa de uma função $g$,
isto é, o conjunto de pontos formado por $g^{-1}(y)$, para $y\in\mathbb{R}$. 
Abaixo vamos mostrar como isto é feito para dois casos particulares de escolhas 
da função $g:\mathbb{R}\to\mathbb{R}$. 

\bigskip 

Seja $g:\mathbb{R}\to\mathbb{R}$ uma função afim da forma $g(x)=ax+b$. 
Considere \(Y \equiv aX + b\) com \(a \neq 0\). Neste caso $g:\mathbb{R}\to\mathbb{R}$ é uma bijeção
e consequentemente uma função invertível. Assim, a imagem inversa de qualquer ponto é um conjunto unitário e neste caso temos 
\[
\mathbb{P}(Y = y) = \mathbb{P}(aX + b = y) = \mathbb{P}\left(X = \frac{1}{a}(y - b)\right)
\quad \text{para qualquer } y \in \mathbb{R}.
\]


Vamos considerar agora outro exemplo. Seja $g:\mathbb{R}\to\mathbb{R}$ dada por
$g(x)=x^2$ e considere a variável aleatória \(Y = X^2\). Para todo $y>0$
temos que $g^{-1}(y) = \{-\sqrt{y}, \sqrt{y}\}$, para $y=0$ temos $g^{-1}(0)=\{0\}$ e para todo $y<0$
temos $g^{-1}(y)=\emptyset$. Logo, segue da expressão \eqref{eq:funcao-massa-g} que
\[
\mathbb{P}(Y = y) = \begin{cases}
\mathbb{P}\bigl(X = \sqrt{y}\bigr) + \mathbb{P}\bigl(X = -\sqrt{y}\bigr) & \text{se } y > 0,\\[0.3cm]
\mathbb{P}(X = 0) & \text{se } y = 0,\\[0.3cm]
0 & \text{se } y < 0.
\end{cases}
\]

\begin{exercise}\label{exercise:poisson-seno}
Seja \(X\) uma variável aleatória discreta com distribuição de Poisson de
parâmetro \(\lambda\), e seja \(Y = \bigl|\sin\bigl(\tfrac{1}{2}\pi X\bigr)\bigr|$.
Determine a função de probabilidade de \(Y\).
\end{exercise}

\section{Esperança}
\label{sec:esperanca}

Considere um dado honesto. Se ele for lançado um grande número de vezes, cada
um dos resultados possíveis \(1, 2, \ldots, 6\) apareceria em cerca de um sexto
dos lançamentos, e a média dos números observados seria aproximadamente
\[
\frac{1}{6} \cdot 1 + \frac{1}{6} \cdot 2 + \cdots + \frac{1}{6} \cdot 6
= \frac{7}{2},
\]
que chamamos de \emph{valor médio}. Essa noção de valor médio estende-se
facilmente a distribuições mais gerais da seguinte forma.

\begin{definition}[Esperança]\label{def:esperanca}
Se \(X\) é uma variável aleatória discreta, a \emph{esperança} de \(X\) é
denotada por \(\mathbb{E}(X)\) e definida por
\begin{equation}\label{eq:esperanca}
\mathbb{E}(X) = \sum_{x \in \operatorname{Im} X} x \, \mathbb{P}(X = x)
\end{equation}
sempre que essa soma convergir absolutamente, isto é, sempre que
\[
\sum_{x \in \operatorname{Im} X} |x\, \mathbb{P}(X = x)| < \infty.
\]
\end{definition}

A equação \eqref{eq:esperanca} é frequentemente escrita como
\[
\mathbb{E}(X) = \sum_{x} x \, \mathbb{P}(X = x) = \sum_{x} x \, p_X(x),
\]
e a esperança de \(X\) é frequentemente chamada de \emph{valor esperado} ou
\emph{média} de \(X\). A razão para exigir convergência absoluta em
\eqref{eq:esperanca} é que a imagem \(\operatorname{Im} X\) pode ser um
conjunto infinito, e precisamos que a soma em \eqref{eq:esperanca} assuma o
mesmo valor independentemente da ordem em que somamos seus termos: sem a
convergência absoluta, o rearranjo dos termos de uma série condicionalmente
convergente poderia alterar o valor da soma.

Uma analogia física para \emph{esperança} é a ideia de \emph{centro de
gravidade}. Se massas com pesos \(\pi_1, \pi_2, \ldots\) são colocadas nos
pontos \(x_1, x_2, \ldots\) de \(\mathbb{R}\), então a posição do centro de
gravidade é \(\sum \pi_i x_i / \sum \pi_i\), ou \(\sum x_i p_i\), onde
\(p_i = \pi_i / \sum_j \pi_j\) é a proporção do peso total alocada à posição
\(x_i\).

Se \(X\) é uma variável aleatória discreta (em algum espaço de probabilidade)
e \(g : \mathbb{R} \to \mathbb{R}\), então \(Y = g(X)\) também é uma variável
aleatória discreta. De acordo com a definição acima, precisamos conhecer a
função de probabilidade de \(Y\) antes de calcular sua esperança. O seguinte
teorema fornece uma maneira útil de evitar esse cálculo tedioso.

\begin{theorem}[Lei do estatístico subconsciente]\label{thm:lotus}
Se \(X\) é uma variável aleatória discreta e \(g : \mathbb{R} \to \mathbb{R}\),
então
\[
\mathbb{E}(g(X)) = \sum_{x \in \operatorname{Im} X} g(x) \, \mathbb{P}(X = x),
\]
sempre que essa soma convergir absolutamente.
\end{theorem}

Intuitivamente, esse resultado é bastante claro, pois \(g(X)\) assume o valor
\(g(x)\) quando \(X\) assume o valor \(x\), um evento com probabilidade
\(\mathbb{P}(X = x)\). Uma prova mais formal é a seguinte.

\begin{proof}
Escrevendo \(I\) para a imagem de \(X\), temos que \(Y = g(X)\) tem imagem
\(g(I)\). Assim,
\[
\begin{aligned}
\mathbb{E}(Y) &= \sum_{y \in g(I)} y \, \mathbb{P}(Y = y) \\
&= \sum_{y \in g(I)} y \sum_{\substack{x \in I \\ g(x) = y}} \mathbb{P}(X = x)
   \quad \text{pela equação \eqref{eq:funcao-massa-g}} \\
&= \sum_{x \in I} g(x) \, \mathbb{P}(X = x)
\end{aligned}
\]
se a última soma convergir absolutamente.
\end{proof}

Duas propriedades simples, mas úteis, da esperança são as seguintes.

\begin{theorem}\label{thm:propriedades-esperanca}
Seja \(X\) uma variável aleatória discreta e sejam \(a, b \in \mathbb{R}\).
\begin{enumerate}[label=(\alph*)]
\item Se \(\mathbb{P}(X \geq 0) = 1\) e \(\mathbb{E}(X) = 0\), então
\(\mathbb{P}(X = 0) = 1\).
\item Temos que \(\mathbb{E}(aX + b) = a\,\mathbb{E}(X) + b\).
\end{enumerate}
\end{theorem}

\begin{proof}
\begin{enumerate}[label=(\alph*)]
\item Suponha que as hipóteses sejam satisfeitas. Como
\(\mathbb{P}(X \geq 0) = 1\), todos os termos
\(x\,\mathbb{P}(X = x)\), com \(x \in \operatorname{Im} X\), são não
negativos. Pela definição \eqref{eq:esperanca} de \(\mathbb{E}(X)\), a soma
desses termos é nula; logo, cada termo deve ser nulo. Portanto,
\(\mathbb{P}(X = x) = 0\) para \(x \neq 0\), e a afirmação segue.
\item Esta é uma consequência simples do \Cref{thm:lotus} aplicado a
\(g(x) = ax + b\).
\end{enumerate}
\end{proof}

Vejamos um exemplo de aplicação do \Cref{thm:lotus}.

\begin{example}\label{ex:poisson-exponencial}
Suponha que \(X\) seja uma variável aleatória com distribuição de Poisson de
parâmetro \(\lambda\), e desejamos encontrar o valor esperado de
\(Y = e^X\). Sem o \Cref{thm:lotus}, teríamos que encontrar a função de
probabilidade de \(Y\). Na verdade, isso não é difícil, mas é ainda mais fácil
aplicar o teorema para obter
\[
\begin{aligned}
\mathbb{E}(Y) &= \mathbb{E}(e^X) \\
&= \sum_{k=0}^{\infty} e^k \, \mathbb{P}(X = k)
= \sum_{k=0}^{\infty} e^k \, \frac{1}{k!} \, \lambda^k \, e^{-\lambda} \\
&= e^{-\lambda} \sum_{k=0}^{\infty} \frac{1}{k!} \, (\lambda e)^k
= e^{\lambda(e - 1)}.
\end{aligned}
\]
% fim do exemplo
\end{example}

\bigskip

A esperança \(\mathbb{E}(X)\) de uma variável aleatória discreta \(X\) é uma
indicação do ``centro'' da distribuição de \(X\). Outra quantidade importante
associada a \(X\) é a \emph{variância} de \(X\), que é uma medida do grau de
dispersão de \(X\) em torno de sua esperança \(\mathbb{E}(X)\).

\begin{definition}[Variância]\label{def:variancia}
A \emph{variância} \(\operatorname{var}(X)\) de uma variável aleatória discreta
\(X\) é definida por
\begin{equation}\label{eq:variancia-def}
\operatorname{var}(X) = \mathbb{E}\left(\left[ X - \mathbb{E}(X) \right]^2\right).
\end{equation}
\end{definition}

Observamos que, pelo \Cref{thm:lotus},
\begin{equation}\label{eq:variancia-soma}
\operatorname{var}(X) = \sum_{x \in \operatorname{Im} X} (x - \mu)^2 \, \mathbb{P}(X = x),
\end{equation}
onde \(\mu = \mathbb{E}(X)\). Uma motivação aproximada para essa definição é a
seguinte. Se a dispersão de \(X\) em torno de sua esperança é muito pequena,
então \(|X - \mu|\) tende a ser pequeno, fazendo com que
\(\operatorname{var}(X) = \mathbb{E}(|X - \mu|^2)\) também seja pequena; por
outro lado, se há frequentemente uma diferença considerável entre \(X\) e sua
média, então \(|X - \mu|\) pode ser grande, fazendo com que
\(\operatorname{var}(X)\) também seja grande.

A equação \eqref{eq:variancia-soma} não é sempre a maneira mais conveniente de
calcular a variância de uma variável aleatória discreta. Podemos expandir o
termo \((x - \mu)^2\) em \eqref{eq:variancia-soma} para obter
\[
\begin{aligned}
\operatorname{var}(X) &= \sum_{x} \left(x^2 - 2\mu x + \mu^2\right) \mathbb{P}(X = x) \\
&= \sum_{x} x^2 \, \mathbb{P}(X = x)
- 2\mu \sum_{x} x \, \mathbb{P}(X = x)
+ \mu^2 \sum_{x} \mathbb{P}(X = x) \\
&= \mathbb{E}(X^2) - 2\mu^2 + \mu^2
   \quad \text{pela definição \eqref{eq:esperanca} e pelo fato de que}
   \quad \sum_{x} \mathbb{P}(X = x) = 1 \\
&= \mathbb{E}(X^2) - \mu^2,
\end{aligned}
\]
onde \(\mu = \mathbb{E}(X)\) como antes. Assim, obtemos a fórmula útil
\begin{equation}\label{eq:variancia-formula}
\operatorname{var}(X) = \mathbb{E}(X^2) - \mathbb{E}(X)^2.
\end{equation}

\begin{example}\label{ex:geometrica}
Se \(X\) tem distribuição geométrica com parâmetro \(p\) (\(= 1 - q\)), a média
de \(X\) é
\[
\begin{aligned}
\mathbb{E}(X) &= \sum_{k=1}^{\infty} k \, p \, q^{k-1} \\
&= \frac{p}{(1 - q)^2} = \frac{1}{p},
\end{aligned}
\]
e a variância de \(X\) é
\[
\operatorname{var}(X) = \sum_{k=1}^{\infty} k^2 \, p \, q^{k-1} - \frac{1}{p^2}
\]
pela fórmula \eqref{eq:variancia-formula}. Agora,
\[
\begin{aligned}
\sum_{k=1}^{\infty} k^2 \, q^{k-1}
&= q \sum_{k=1}^{\infty} k(k-1) \, q^{k-2} + \sum_{k=1}^{\infty} k \, q^{k-1} \\
&= \frac{2q}{(1 - q)^3} + \frac{1}{(1 - q)^2},
\end{aligned}
\]
usando a técnica de diferenciação de séries de potências
\footnote{Para somar uma série como \(\sum_{k=0}^{\infty} k x^{k-1}\), basta
observar que, se \(|x| < 1\), então
\(\sum_k k x^{k-1} = \tfrac{d}{dx} \sum_k x^k\) e, portanto,
\(\sum_{k=0}^{\infty} k x^{k-1} = \tfrac{d}{dx} (1-x)^{-1} = (1-x)^{-2}\).
A propriedade relevante das séries de potências é que elas podem ser
diferenciadas termo a termo dentro de seu círculo de convergência;
diferenciações repetidas de \((1-x)^{-1}\) fornecem fórmulas para
\(\sum_k k(k-1) x^{k-2}\) e expressões análogas.}, o que dá
\[
\begin{aligned}
\operatorname{var}(X) &= p \left(\frac{2q}{p^3} + \frac{1}{p^2}\right) - \frac{1}{p^2} \\
&= q \, p^{-2}.
\end{aligned}
\]
% fim do exemplo
\end{example}

\begin{exercise}\label{exercise:binomial-variancia}
Se \(X\) tem distribuição binomial com parâmetros \(n\) e \(p = 1 - q\), mostre
que
\[
\mathbb{E}(X) = np, \qquad \mathbb{E}(X^2) = npq + n^2 p^2,
\]
e deduza a variância de \(X\).
\end{exercise}

\begin{exercise}\label{exercise:var-afim}
Mostre que \(\operatorname{var}(aX + b) = a^2 \operatorname{var}(X)\) para
\(a, b \in \mathbb{R}\).
\end{exercise}

\begin{exercise}\label{exercise:poisson-variancia}
Encontre \(\mathbb{E}(X)\) e \(\mathbb{E}(X^2)\) quando \(X\) tem distribuição
de Poisson com parâmetro \(\lambda\), e mostre que a distribuição de Poisson
tem variância igual à sua média.
\end{exercise}
